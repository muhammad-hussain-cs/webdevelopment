// ==================== JAVASCRIPT (Vanilla JS) ====================
// 1. INPUT VALIDATION (2 Marks) + 2. DYNAMIC MESSAGE (2 Marks)
document.getElementById('registrationForm').addEventListener('submit', function(e) {
  e.preventDefault();
  var name = document.getElementById('studentName').value.trim();
  var email = document.getElementById('studentEmail').value.trim();
  var eventName = document.getElementById('eventSelect').value;
  var msg = document.getElementById('messageArea');
  var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  if (name === '' || email === '') {
    msg.innerHTML = '<div class="alert alert-danger"><strong>❌ Error:</strong> Please fill in both your Name and Email. Required fields cannot be empty.</div>';
    return;
  }
  if (!emailPattern.test(email)) {
    msg.innerHTML = '<div class="alert alert-danger"><strong>❌ Error:</strong> Please enter a valid email address, e.g. name@university.edu.pk</div>';
    return;
  }
  if (eventName === '') {
    msg.innerHTML = '<div class="alert alert-danger"><strong>❌ Error:</strong> Please select an event from the dropdown.</div>';
    return;
  }
  // Success - personalised dynamic message
  msg.innerHTML = '<div class="alert alert-success"><strong>🎉 Registration Successful!</strong><br>Welcome, <strong>' + name + '</strong>! You have successfully registered for <strong>' + eventName + '</strong>. A confirmation will be sent to <strong>' + email + '</strong>.</div>';
});

// 3. STYLE MANIPULATION (1 Mark) - Vanilla JS, cycles card theme
var themes = ['theme-1', 'theme-2', 'theme-3', 'theme-4'];
var themeIndex = 0;
document.getElementById('themeBtn').addEventListener('click', function() {
  var card = document.getElementById('registrationCard');
  card.classList.remove(themes[themeIndex]);
  themeIndex = (themeIndex + 1) % themes.length;
  card.classList.add(themes[themeIndex]);
  document.getElementById('mainTitle').style.color = themeIndex === 1 ? '#fbbf24' : '';
});

// Reset button (Vanilla JS)
document.getElementById('resetBtn').addEventListener('click', function() {
  document.getElementById('registrationForm').reset();
  document.getElementById('messageArea').innerHTML = '';
});

// ==================== jQUERY ====================
$(document).ready(function() {
  // 1. SHOW / HIDE (2 Marks) - $("#id") selector + click()
  $('#toggleInstructionsBtn').click(function() {
    $('#instructions').toggle();
    $(this).text($(this).text().indexOf('Hide') === 0 ? 'Show Instructions' : 'Hide Instructions');
  });

  // 2. jQuery EFFECT (2 Marks) - slideToggle()
  $('#toggleDetailsBtn').click(function() {
    $('#eventDetails').slideToggle(400);
    $(this).text($(this).text().indexOf('Show') === 0 ? 'Hide Event Details ▲' : 'Show Event Details ▼');
  });

  // 3. DYNAMIC CONTENT (2 Marks) - add to list + update element with jQuery
  var count = 0;
  $('#addEventBtn').click(function() {
    var name = $('#studentName').val().trim();
    var eventName = $('#eventSelect').val();
    if (name === '' || eventName === '') {
      $('#messageArea').html('<div class="alert alert-warning"><strong>⚠️ Please enter your name and select an event first, then click “Add to My Events”.</strong></div>');
      return;
    }
    if (count === 0) { $('#myEventsList').empty(); }
    count++;
    $('#myEventsList').append('<li class="list-group-item event-item">✅ <strong>' + eventName + '</strong> — ' + name + '</li>');
    $('#eventCount').text(count);
    $('#messageArea').html('<div class="alert alert-info"><strong>Added!</strong> ' + eventName + ' was added to your list. Total events: ' + count + '</div>');
    // fadeIn effect on the new item
    $('#myEventsList li:last').hide().fadeIn(600);
  });

  // 4. EVENT HANDLING & SELECTORS (2 Marks) - mouseover() + $(".class") selector
  $('.event-item').mouseover(function() {
    $(this).css('background-color', '#dbeafe');
  });
  // For dynamically added items, use delegation
  $(document).on('mouseover', '.event-item', function() {
    $(this).css('background-color', '#dbeafe');
  });
  $(document).on('mouseout', '.event-item', function() {
    $(this).css('background-color', '');
  });
  // mouseover on heading changes its appearance
  $('#mainTitle').mouseover(function() {
    $(this).css('letter-spacing', '1px');
  });
});
</script>
